export default function LiqPayStandaloneButton({
  onClick,
  loading = false,
  disabled = false,
}: LiqPayStandaloneButtonProps) {
  const [visible, setVisible] = useState(false)
 
  useEffect(() => {
    const timer = setTimeout(() => setVisible(true), 100)
    return () => clearTimeout(timer)
  }, [])
 
  useEffect(() => {
    const styleEl = document.createElement('style')
    styleEl.textContent = liqpayStyles
    document.head.appendChild(styleEl)
    return () => styleEl.remove()
  }, [])
 
  return (
    <Button
      className="h-[58px] min-w-[320px] w-full max-w-[320px] gap-2 overflow-visible"
      onClick={onClick}
      disabled={disabled || loading}
      type="button"
    >
      {loading ? (
        <span className="liqpay-btn--spinner" />
      ) : (
        <div className="liqpay-btn--inner">
          <div
            className="liqpay-btn--logo-wrapper"
            style={{
              opacity: visible ? 1 : 0,
              transition: 'opacity 0.5s ease',
            }}
          >
            <LiqPayLogoSvg />
          </div>
        </div>
      )}
    </Button>
  )
}
